# Replacement CMakeLists.txt for the stag vcpkg port.
# Upstream builds an example executable and has no install rules,
# this builds only the library and exports a cmake package config.
cmake_minimum_required(VERSION 3.16)
project(stag CXX)

find_package(OpenCV REQUIRED)

file(GLOB STAG_SOURCES "src/*.cpp" "src/ED/*.cpp")
add_library(stag ${STAG_SOURCES})

target_include_directories(stag
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/src"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/ED"
    INTERFACE
        $<INSTALL_INTERFACE:include>
)
# OpenCV_LIBS resolves to the module targets, or opencv_world on world builds
target_link_libraries(stag PUBLIC ${OpenCV_LIBS})
target_compile_features(stag PRIVATE cxx_std_11)
set_target_properties(stag PROPERTIES POSITION_INDEPENDENT_CODE ON)

include(GNUInstallDirs)
install(TARGETS stag EXPORT stag-targets
    RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}"
    LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
    ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
)
# only the public detection header is installed, internals stay private
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/src/Stag.h"
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/stag"
)
install(EXPORT stag-targets
    NAMESPACE stag::
    DESTINATION "lib/cmake/stag"
    FILE "stag-targets.cmake"
)

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stag-config.cmake"
[[include(CMakeFindDependencyMacro)
find_dependency(OpenCV)
include("${CMAKE_CURRENT_LIST_DIR}/stag-targets.cmake")
]])
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/stag-config.cmake"
    DESTINATION "lib/cmake/stag"
)
